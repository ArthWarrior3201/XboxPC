cmake_minimum_required(VERSION 3.20)
project(XboxPC VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(xboxpc_host SHARED runtime/host.cpp)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "XboxPC requires x64")
  endif()
  target_sources(xboxpc_host PRIVATE runtime/d3d12.cpp)
  target_link_libraries(xboxpc_host PRIVATE d3d12 dxgi xinput9_1_0 user32)
  add_library(xboxpc_guest SHARED runtime/guest.cpp)
  add_executable(xboxpc_fixture runtime/demo.cpp)
  target_link_libraries(xboxpc_fixture PRIVATE xboxpc_guest)
endif()
add_executable(runtime_test runtime/runtime_test.cpp)
target_link_libraries(runtime_test PRIVATE xboxpc_host)
add_test(NAME runtime_conformance COMMAND runtime_test)
